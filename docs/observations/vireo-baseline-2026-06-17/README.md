```bash
# Counts
wc -l state/telemetry/*.jsonl
wc -l <new-run>/telemetry/*.jsonl

# Agent errors breakdown
cat state/telemetry/agent-errors.jsonl | python3 -c "
import sys, json
from collections import Counter
errs = [json.loads(l) for l in sys.stdin]
print('Errors:', Counter(e.get('error_class') for e in errs))
print('Phases:', Counter(e.get('phase') for e in errs))
"

# Budget per task
cat state/telemetry/budget-usage.jsonl | python3 -c "
import sys, json
events = [json.loads(l) for l in sys.stdin if json.loads(l).get('event') == 'task_complete']
print(f'Tasks: {len(events)}, total tokens: {sum(e[\"tokens\"] for e in events):,}')
"

# Review request-changes rate
grep -c "request-changes" state/telemetry/agent-errors.jsonl

# next-env trip count (should be 0 after fix)

```
